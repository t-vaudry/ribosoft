29 public async Task InvokeAsync(HttpContext context, IActivityLogService activityLogService)
30 {
31 // Skip logging for static files and health checks

34 await _next(context);

58 private static bool ShouldSkipLogging(PathString path)
59 {
60 var pathValue = path.Value?.ToLower() ?? "";
61
62 return pathValue.StartsWith("/css/") ||
63 pathValue.StartsWith("/js/") ||
64 pathValue.StartsWith("/images/") ||
65 pathValue.StartsWith("/lib/") ||
66 pathValue.StartsWith("/favicon.ico") ||
67 pathValue.StartsWith("/health") ||
68 pathValue.StartsWith("/hangfire/") ||
69 pathValue.Contains(".map") ||
70 pathValue.Contains(".woff") ||
71 pathValue.Contains(".ttf") ||
72 pathValue.Contains(".eot");
73 }

80 private async Task LogRequestAsync(HttpContext context, IActivityLogService activityLogService, long duration)
81 {
82 try
83 {
84 var request = context.Request;
85 var response = context.Response;
86 var user = context.User;
87
88 var userId = user.FindFirst(ClaimTypes.NameIdentifier)?.Value;
89 var userName = user.FindFirst(ClaimTypes.Name)?.Value ?? user.Identity?.Name;
90 var ipAddress = GetClientIpAddress(context);
91
92 var logLevel = response.StatusCode >= 500 ? "Error" :
93 response.StatusCode >= 400 ? "Warning" : "Information";
94
95 var category = DetermineCategory(request.Path);
96 var message = $"{request.Method} {request.Path} - {response.StatusCode}";
97
98 await activityLogService.LogAsync(
99 logLevel: logLevel,
100 category: category,
101 message: message,
102 userId: userId,
103 userName: userName,
104 ipAddress: ipAddress,
105 requestPath: request.Path,
106 requestMethod: request.Method,
107 statusCode: response.StatusCode,
108 duration: duration
109 );
110 }
111 catch (Exception ex)
112 {
113 _logger.LogError(ex, "Failed to log HTTP request");
114 }
115 }

121 private static string? GetClientIpAddress(HttpContext context)
122 {
123 // Check for forwarded IP first (for load balancers/proxies)
124 var forwardedFor = context.Request.Headers["X-Forwarded-For"].FirstOrDefault();
125 if (!string.IsNullOrEmpty(forwardedFor))
126 {
127 return forwardedFor.Split(',')[0].Trim();
128 }
129
130 var realIp = context.Request.Headers["X-Real-IP"].FirstOrDefault();
131 if (!string.IsNullOrEmpty(realIp))
132 {
133 return realIp;
134 }
135
136 return context.Connection.RemoteIpAddress?.ToString();
137 }

143 private static string DetermineCategory(PathString path)
144 {
145 var pathValue = path.Value?.ToLower() ?? "";
146
147 return pathValue switch
148 {
149 var p when p.StartsWith("/account/") => "Authentication",
150 var p when p.StartsWith("/admin/") => "UserManagement",
151 var p when p.StartsWith("/jobs/") => "JobExecution",
152 var p when p.StartsWith("/request/") => "JobSubmission",
153 var p when p.StartsWith("/ribozymes/") => "RibozymeDesign",
154 var p when p.StartsWith("/designs/") => "RibozymeDesign",
155 var p when p.StartsWith("/api/") => "API",
156 var p when p.StartsWith("/manage/") => "UserManagement",
157 _ => "System"
158 };
159 }
