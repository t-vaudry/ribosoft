159 private static void ConfigureServices(IServiceCollection services, IConfiguration configuration)
160 {
161 // Database configuration
162 var providerName = configuration.GetValue("EntityFrameworkProvider", "SqlServer");
163 var connectionString = configuration.GetConnectionString($"{providerName}Connection");
164
165 if (string.IsNullOrEmpty(connectionString))
166 {
167 throw new ArgumentException("Connection string cannot be empty.");
168 }
169
170 if (providerName == "Npgsql")
171 {
172 services.AddDbContext<ApplicationDbContext>(options =>
173 options.UseNpgsql(connectionString));
174
175 services.AddHangfire(x => x
176 .UsePostgreSqlStorage(options =>
177 {
178 options.UseNpgsqlConnection(connectionString);
179 }, new PostgreSqlStorageOptions

182 }));
183 }
184 else if (providerName == "SqlServer")

186 services.AddDbContext<ApplicationDbContext>(options =>
187 options.UseSqlServer(connectionString));

195 }

197 // Identity configuration
