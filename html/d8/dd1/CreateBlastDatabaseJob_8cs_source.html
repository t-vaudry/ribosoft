43 [AutomaticRetry(Attempts = 2, DelaysInSeconds = new[] { 300, 900 })]
44 public async Task CreateBlastDatabaseAsync(int downloadId, IJobCancellationToken cancellationToken)
45 {
46 var download = await _context.DatasetDownloads.FindAsync(downloadId);
47 if (download == null)
48 {
49 _logger.LogError("Download record {DownloadId} not found", downloadId);
50 return;
51 }
52
53 if (download.Status != DatasetDownloadStatus.Completed)
54 {
55 _logger.LogWarning("Download {DownloadId} is not completed, cannot create BLAST database", downloadId);
56 return;
57 }
58
59 try
60 {
61 _logger.LogInformation("Starting BLAST database creation for dataset {AccessionId} (ID: {DownloadId})",
62 download.AccessionId, downloadId);
63
64 // Update status
65 download.Status = DatasetDownloadStatus.Processing;
66 await _context.SaveChangesAsync();
67
68 cancellationToken.ThrowIfCancellationRequested();
69
70 // Step 1: Extract the downloaded ZIP file
71 var extractPath = await ExtractDataset(download, cancellationToken);
72
73 // Step 2: Find FASTA files in the extracted content
74 var fastaFiles = FindFastaFiles(extractPath);
75 if (fastaFiles.Count == 0)
76 {
77 throw new InvalidOperationException($"No FASTA files found in dataset {download.AccessionId}");
78 }
79
80 // Step 3: Create BLAST database for each FASTA file (if enabled)
81 var blastDbPath = _configuration["Assemblies:Path"] ??
82 Path.Combine(Directory.GetCurrentDirectory(), "BlastDatabases");
83 blastDbPath = ExpandPath(blastDbPath);
84 Directory.CreateDirectory(blastDbPath);
85
86 var createBlastDb = _configuration.GetValue<bool>("Assemblies:AutoCreateBlastDatabase", true);
87
88 if (!createBlastDb)
89 {
90 _logger.LogInformation("BLAST database creation is disabled, skipping for {AccessionId}", download.AccessionId);
91 }
92 else
93 {
94 // Check if BLAST database creation should be skipped for large files
95 var maxFileSizeMB = _configuration.GetValue<double>("Assemblies:MaxBlastDbFileSizeMB", 500);
96 var skipLargeFiles = _configuration.GetValue<bool>("Assemblies:SkipBlastDbForLargeFiles", false);
97
98 var processedFiles = 0;
99 var skippedFiles = 0;
100 var totalFiles = fastaFiles.Count;
101
102 _logger.LogInformation("Starting BLAST database creation for {TotalFiles} FASTA files", totalFiles);
103
104 foreach (var fastaFile in fastaFiles)
105 {
106 cancellationToken.ThrowIfCancellationRequested();
107
108 var fileInfo = new FileInfo(fastaFile);
109 var fileSizeMB = fileInfo.Length / (1024.0 * 1024.0);
110
111 if (skipLargeFiles && fileSizeMB > maxFileSizeMB)
112 {
113 _logger.LogWarning("Skipping BLAST database creation for large file {FileName} ({SizeMB:F1} MB > {MaxSizeMB} MB limit)",
114 Path.GetFileName(fastaFile), fileSizeMB, maxFileSizeMB);
115 skippedFiles++;
116 continue;
117 }
118
119 // Update progress
120 var progressPercent = 70 + (processedFiles * 20 / totalFiles); // 70-90% range for BLAST creation
121 await UpdateDownloadProgress(download, progressPercent,
122 $"Creating BLAST database {processedFiles + 1}/{totalFiles}: {Path.GetFileName(fastaFile)}");
123
124 await CreateBlastDatabase(fastaFile, blastDbPath, download.AccessionId, cancellationToken);
125 processedFiles++;
126 }
127
128 _logger.LogInformation("BLAST database creation completed: {ProcessedFiles} processed, {SkippedFiles} skipped",
129 processedFiles, skippedFiles);
130 }
131
132 // Step 4: Create Assembly record for integration with existing system
133 await CreateAssemblyRecord(download, blastDbPath);
134
135 // Step 5: Clean up temporary files (preserve BLAST databases)
136 var cleanupExtracted = _configuration.GetValue<bool>("Assemblies:CleanupAfterProcessing", true);
137 var cleanupZip = _configuration.GetValue<bool>("Assemblies:CleanupZipFiles", true);
138
139 if (cleanupExtracted)
140 {
141 try
142 {
143 // Verify that extractPath is different from blastDbPath to prevent accidental deletion
144 var normalizedExtractPath = Path.GetFullPath(extractPath);
145 var normalizedBlastDbPath = Path.GetFullPath(Path.Combine(blastDbPath, download.AccessionId));
146
147 if (normalizedExtractPath.Equals(normalizedBlastDbPath, StringComparison.OrdinalIgnoreCase))
148 {
149 _logger.LogWarning("Skipping cleanup: Extract path and BLAST DB path are the same: {Path}", normalizedExtractPath);
150 }
151 else
152 {
153 _logger.LogInformation("Cleaning up temporary extracted files at {ExtractPath} (BLAST DBs preserved at {BlastDbPath})",
154 extractPath, normalizedBlastDbPath);
155
156 Directory.Delete(extractPath, true);
157 _logger.LogInformation("Successfully cleaned up extracted files at {ExtractPath}", extractPath);
158 }
159 }
160 catch (Exception ex)
161 {
162 _logger.LogWarning(ex, "Failed to clean up extracted files at {ExtractPath}", extractPath);
163 }
164 }
165
166 if (cleanupZip && !string.IsNullOrEmpty(download.LocalPath) && File.Exists(download.LocalPath))
167 {
168 try
169 {
170 var zipPath = download.LocalPath;
171 File.Delete(zipPath);
172 _logger.LogInformation("Cleaned up original ZIP file at {ZipPath}", zipPath);
173
174 // Clear the local path since file is deleted
175 download.LocalPath = null;
176 await _context.SaveChangesAsync();
177 }
178 catch (Exception ex)
179 {
180 _logger.LogWarning(ex, "Failed to clean up ZIP file at {ZipPath}", download.LocalPath);
181 }
182 }
183
184 // Step 6: Log processing summary
185 var finalBlastDbPath = Path.Combine(blastDbPath, download.AccessionId);
186 var blastDbFiles = Directory.Exists(finalBlastDbPath) ?
187 Directory.GetFiles(finalBlastDbPath, "*", SearchOption.AllDirectories).Length : 0;
188
189 _logger.LogInformation("Dataset processing completed for {AccessionId}:", download.AccessionId);
190 _logger.LogInformation(" ✅ BLAST databases preserved: {BlastDbPath} ({FileCount} files)", finalBlastDbPath, blastDbFiles);
191 _logger.LogInformation(" 🗑️ Temporary files cleaned: {ExtractPath} (removed)", extractPath);
192 _logger.LogInformation(" 🗑️ Original ZIP cleaned: {ZipPath} ({Status})",
193 download.LocalPath ?? "N/A",
194 cleanupZip ? "removed" : "preserved");
195
196 // Step 7: Complete
197 download.Status = DatasetDownloadStatus.ReadyForBlast;
198 await _context.SaveChangesAsync();
199
200 _logger.LogInformation("Successfully created BLAST database for dataset {AccessionId}",
201 download.AccessionId);
202
203 // Trigger assembly database rescan to pick up the new database
204 BackgroundJob.Enqueue<UpdateAssemblyDatabase>(
205 job => job.Rescan(JobCancellationToken.Null));
206 }
207 catch (OperationCanceledException)
208 {
209 _logger.LogInformation("BLAST database creation cancelled for dataset {AccessionId}", download.AccessionId);
210 download.Status = DatasetDownloadStatus.Cancelled;
211 await _context.SaveChangesAsync();
212 }
213 catch (Exception ex)
214 {
215 _logger.LogError(ex, "Error creating BLAST database for dataset {AccessionId}", download.AccessionId);
216
217 download.Status = DatasetDownloadStatus.Failed;
218 download.ErrorMessage = $"BLAST database creation failed: {ex.Message}";
219 await _context.SaveChangesAsync();
220
221 throw; // Re-throw for Hangfire retry mechanism
222 }

228 private async Task<string> ExtractDataset(DatasetDownload download, IJobCancellationToken cancellationToken)
229 {
230 if (string.IsNullOrEmpty(download.LocalPath))
231 {
232 throw new InvalidOperationException("LocalPath is null or empty for dataset download");
233 }
234
235 var directoryName = Path.GetDirectoryName(download.LocalPath);
236 if (string.IsNullOrEmpty(directoryName))
237 {
238 throw new InvalidOperationException($"Unable to get directory name from LocalPath: {download.LocalPath}");
239 }
240
241 var extractPath = Path.Combine(
242 directoryName,
243 Path.GetFileNameWithoutExtension(download.LocalPath));
244
245 if (Directory.Exists(extractPath))
246 {
247 Directory.Delete(extractPath, true);
248 }
249
250 Directory.CreateDirectory(extractPath);
251
252 _logger.LogInformation("Extracting dataset {AccessionId} to {ExtractPath}",
253 download.AccessionId, extractPath);
254
255 await Task.Run(() =>
256 {
257 if (string.IsNullOrEmpty(download.LocalPath))
258 {
259 throw new InvalidOperationException("LocalPath is null or empty for ZIP extraction");
260 }
261
262 using var archive = ZipFile.OpenRead(download.LocalPath);
263 foreach (var entry in archive.Entries)
264 {
265 cancellationToken.ThrowIfCancellationRequested();
266
267 if (!string.IsNullOrEmpty(entry.Name))
268 {
269 // Security: Prevent Zip Slip attacks by validating the destination path
270 var fullExtractPath = Path.GetFullPath(extractPath);
271
272 // Normalize the entry name to prevent path traversal
273 var normalizedEntryName = entry.FullName.Replace('/', Path.DirectorySeparatorChar)
274 .Replace('\\', Path.DirectorySeparatorChar);
275
276 // Remove any leading path separators or relative path components
277 normalizedEntryName = normalizedEntryName.TrimStart(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar);
278
279 // Reject entries with path traversal attempts
280 if (normalizedEntryName.Contains("..") || Path.IsPathRooted(normalizedEntryName))
281 {
282 _logger.LogWarning("Skipping potentially malicious archive entry - path traversal attempt detected");
283 continue;
284 }
285
286 var destinationPath = Path.Combine(extractPath, normalizedEntryName);
287 var fullDestinationPath = Path.GetFullPath(destinationPath);
288
289 // Final validation: ensure the resolved path is within the extract directory
290 if (!fullDestinationPath.StartsWith(fullExtractPath + Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase) &&
291 !fullDestinationPath.Equals(fullExtractPath, StringComparison.OrdinalIgnoreCase))
292 {
293 _logger.LogWarning("Skipping potentially malicious archive entry - path traversal attempt detected");
294 continue;
295 }
296
297 // Use the validated path for all operations
298 var destinationDir = Path.GetDirectoryName(fullDestinationPath);
299
300 if (!string.IsNullOrEmpty(destinationDir))
301 {
302 Directory.CreateDirectory(destinationDir);
303 }
304
305 entry.ExtractToFile(fullDestinationPath, true);
306 }
307 }
308 });
309
310 return extractPath;
311 }

377 private async Task CreateBlastDatabase(string fastaFile, string blastDbPath, string accessionId, IJobCancellationToken? cancellationToken = null)
378 {
379 var fileName = Path.GetFileNameWithoutExtension(fastaFile);
380 var dbName = $"{accessionId}_{fileName}";
381
382 // Create accession-specific directory for better organization
383 var accessionDbPath = Path.Combine(blastDbPath, accessionId);
384 Directory.CreateDirectory(accessionDbPath);
385
386 var dbPath = Path.Combine(accessionDbPath, dbName);
387
388 // Get file size for progress estimation
389 var fileInfo = new FileInfo(fastaFile);
390 var fileSizeMB = fileInfo.Length / (1024.0 * 1024.0);
391
392 _logger.LogInformation("Creating BLAST database {DbName} from {FastaFile} ({SizeMB:F1} MB) in {AccessionDbPath}",
393 dbName, Path.GetFileName(fastaFile), fileSizeMB, accessionDbPath);
394
395 // Check if database already exists
396 if (File.Exists($"{dbPath}.nhr") && File.Exists($"{dbPath}.nin") && File.Exists($"{dbPath}.nsq"))
397 {
398 _logger.LogInformation("BLAST database {DbName} already exists, skipping creation", dbName);
399 return;
400 }
401
402 var makeblastdbPath = _configuration["Assemblies:MakeBlastDbPath"] ?? "makeblastdb";
403
404 // Determine database type based on file content/name
405 var dbType = DetermineDbType(fastaFile, fileName);
406
407 var processInfo = new ProcessStartInfo
408 {
409 FileName = makeblastdbPath,
410 Arguments = $"-in \"{fastaFile}\" -dbtype {dbType} -out \"{dbPath}\" -title \"{dbName}\" -parse_seqids",
411 UseShellExecute = false,
412 RedirectStandardOutput = true,
413 RedirectStandardError = true,
414 CreateNoWindow = true
415 };
416
417 using var process = new Process { StartInfo = processInfo };
418
419 // Set up timeout based on file size (rough estimate: 1 minute per 10MB + 5 minute base)
420 var timeoutMinutes = Math.Max(5, (int)(fileSizeMB / 10) + 5);
421 var timeout = TimeSpan.FromMinutes(timeoutMinutes);
422
423 _logger.LogInformation("Starting makeblastdb for {DbName} with {TimeoutMinutes} minute timeout",
424 dbName, timeoutMinutes);
425
426 var startTime = DateTime.UtcNow;
427 process.Start();
428
429 // Monitor process with cancellation and timeout
430 var outputTask = process.StandardOutput.ReadToEndAsync();
431 var errorTask = process.StandardError.ReadToEndAsync();
432
433 try
434 {
435 // Create a combined cancellation token for timeout and job cancellation
436 using var timeoutCts = new CancellationTokenSource(timeout);
437 using var combinedCts = CancellationTokenSource.CreateLinkedTokenSource(
438 timeoutCts.Token,
439 cancellationToken?.ShutdownToken ?? CancellationToken.None);
440
441 // Wait for process completion with cancellation
442 await process.WaitForExitAsync(combinedCts.Token);
443
444 var output = await outputTask;
445 var error = await errorTask;
446 var duration = DateTime.UtcNow - startTime;
447
448 if (process.ExitCode != 0)
449 {
450 throw new InvalidOperationException(
451 $"makeblastdb failed for {dbName} with exit code {process.ExitCode} after {duration:mm\\:ss}. " +
452 $"Error: {error}. Output: {output}");
453 }
454
455 _logger.LogInformation("Successfully created BLAST database {DbName} in {Duration:mm\\:ss} at {DbPath}",
456 dbName, duration, accessionDbPath);
457 }
458 catch (OperationCanceledException) when (cancellationToken?.ShutdownToken.IsCancellationRequested == true)
459 {
460 _logger.LogWarning("BLAST database creation for {DbName} was cancelled by job cancellation", dbName);
461
462 if (!process.HasExited)
463 {
464 process.Kill(true);
465 await process.WaitForExitAsync();
466 }
467 throw;
468 }
469 catch (OperationCanceledException)
470 {
471 var duration = DateTime.UtcNow - startTime;
472 _logger.LogError("BLAST database creation for {DbName} timed out after {Duration:mm\\:ss} (limit: {TimeoutMinutes} minutes)",
473 dbName, duration, timeoutMinutes);
474
475 if (!process.HasExited)
476 {
477 process.Kill(true);
478 await process.WaitForExitAsync();
479 }
480
481 throw new TimeoutException(
482 $"BLAST database creation for {dbName} timed out after {duration:mm\\:ss}. " +
483 $"Large datasets may require more time. Consider increasing the timeout or processing smaller files.");
484 }
485 }

503 private static string ExpandPath(string path)
504 {
505 if (string.IsNullOrEmpty(path))
506 return path;
507
508 // Expand ~ to home directory
509 if (path.StartsWith("~/") || path == "~")
510 {
511 var homeDir = Environment.GetFolderPath(Environment.SpecialFolder.UserProfile);
512 if (path == "~")
513 return homeDir;

597 private string DetermineAssemblyTypeFromFiles(string assemblyPath)
598 {
599 if (string.IsNullOrEmpty(assemblyPath) || !Directory.Exists(assemblyPath))
600 {
601 return "Unknown";
602 }
603
604 try
605 {
606 var files = Directory.GetFiles(assemblyPath, "*", SearchOption.TopDirectoryOnly)
607 .Select(f => Path.GetFileName(f).ToLowerInvariant())
608 .ToList();
609
610 // Count different types of BLAST databases
611 var hasGenome = files.Any(f => f.Contains("genomic") && (f.EndsWith(".nhr") || f.EndsWith(".phr")));
612 var hasProtein = files.Any(f => f.Contains("protein") && f.EndsWith(".phr"));
613 var hasRNA = files.Any(f => f.Contains("rna") && f.EndsWith(".nhr"));
614 var hasCDS = files.Any(f => f.Contains("cds") && f.EndsWith(".nhr"));
615
616 // Determine primary type based on available databases
617 var types = new List<string>();
618
619 if (hasGenome) types.Add("Genome");
620 if (hasProtein) types.Add("Proteome");
621 if (hasRNA) types.Add("Transcriptome");
622 if (hasCDS) types.Add("CDS");
623
624 if (types.Count == 0)
625 {
626 // Fallback: check for any BLAST database files
627 var hasAnyBlastDb = files.Any(f => f.EndsWith(".nhr") || f.EndsWith(".phr") ||
628 f.EndsWith(".nin") || f.EndsWith(".pin"));
629 return hasAnyBlastDb ? "Assembly" : "Unknown";
630 }
631
632 // Return combined type or primary type
633 if (types.Count == 1)
634 {
635 return types[0];
636 }
637 else if (types.Contains("Genome"))
638 {
639 // If genome is present with others, it's likely a complete genome assembly
640 return "Complete Genome";
641 }
642 else
643 {
644 // Multiple types without genome
645 return string.Join(" + ", types);
646 }
647 }
648 catch (Exception ex)
649 {
650 _logger.LogWarning(ex, "Failed to determine assembly type for path: {Path}", assemblyPath);
651 return "Unknown";
652 }
653 }
