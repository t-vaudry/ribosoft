71 public async Task<IActionResult> Index()
72 {
73 ViewBag.BlastDbPath = _configuration["Assemblies:Path"] ?? "Not configured";
74
75 // Get current assemblies and recent downloads
76 var assemblies = await _context.Assemblies.ToListAsync();
77 var recentDownloads = await _context.DatasetDownloads
78 .OrderByDescending(d => d.CreatedAt)
79 .Take(10)
80 .ToListAsync();
81
82 // Process assemblies to fix missing data and calculate sizes
83 foreach (var assembly in assemblies)
84 {
85 _logger.LogInformation("Processing assembly {AccessionId}: Type='{Type}', Path='{Path}'",
86 assembly.AccessionId, assembly.Type, assembly.Path);
87
88 // Fix missing or incorrect Path FIRST
89 var originalPath = assembly.Path;
90 if (string.IsNullOrEmpty(assembly.Path) || assembly.Path.StartsWith("~/"))
91 {
92 var blastDbPath = _configuration["Assemblies:Path"] ??
93 Path.Combine(Directory.GetCurrentDirectory(), "BlastDatabases");
94 blastDbPath = ExpandPath(blastDbPath);
95
96 // Use organized path structure
97 assembly.Path = Path.Combine(blastDbPath, assembly.AccessionId);
98 _logger.LogInformation("Updated path for {AccessionId}: '{OldPath}' -> '{NewPath}'",
99 assembly.AccessionId, originalPath, assembly.Path);
100 }
101
102 // Fix missing Type - determine actual assembly type (now with correct path)
103 if (string.IsNullOrEmpty(assembly.Type) || assembly.Type == "Downloaded")
104 {
105 assembly.Type = DetermineAssemblyType(assembly.Path, assembly.AccessionId);
106 _logger.LogInformation("Determined type for {AccessionId}: {Type}", assembly.AccessionId, assembly.Type);
107 }
108
109 // Calculate size
110 assembly.Size = CalculateAssemblySize(assembly.Path);
111 _logger.LogInformation("Assembly {AccessionId} final: Type='{Type}', Path='{Path}', Size={Size}",
112 assembly.AccessionId, assembly.Type, assembly.Path, assembly.Size);
113 }
114
115 ViewBag.RecentDownloads = recentDownloads;
116
117 return View(assemblies);
118 }

467 [HttpPost]
468 public async Task<IActionResult> Delete(int id)
469 {
470 _logger.LogInformation("Delete assembly with taxonomy ID: {TaxonomyId} by user: {User}", id, User.Identity?.Name);
471
472 try
473 {
474 var assembly = await _context.Assemblies.FirstOrDefaultAsync(a => a.TaxonomyId == id);
475 if (assembly == null)
476 {
477 return Json(new { success = false, message = "Assembly not found" });
478 }
479
480 // Check if assembly is being used by any active jobs
481 var activeJobsCount = await _context.Jobs
482 .Where(j => j.AssemblyId == id &&
483 (j.JobState == JobState.New || j.JobState == JobState.Started ||
484 j.JobState == JobState.CandidateGenerator || j.JobState == JobState.Structure ||
485 j.JobState == JobState.MultiObjectiveOptimization || j.JobState == JobState.Specificity ||
486 j.JobState == JobState.QueuedPhase2 || j.JobState == JobState.QueuedPhase3))
487 .CountAsync();
488
489 if (activeJobsCount > 0)
490 {
491 return Json(new {
492 success = false,
493 message = $"Cannot delete assembly. It is currently being used by {activeJobsCount} active job(s)."
494 });
495 }
496
497 // Delete files from filesystem
498 var deletedSize = 0L;
499 if (!string.IsNullOrEmpty(assembly.Path) && Directory.Exists(assembly.Path))
500 {
501 try
502 {
503 deletedSize = CalculateAssemblySize(assembly.Path);
504 Directory.Delete(assembly.Path, true);
505 _logger.LogInformation("Deleted assembly directory: {Path} ({Size} bytes)", assembly.Path, deletedSize);
506 }
507 catch (Exception ex)
508 {
509 _logger.LogWarning(ex, "Failed to delete assembly directory: {Path}", assembly.Path);
510 return Json(new {
511 success = false,
512 message = $"Failed to delete assembly files from filesystem: {ex.Message}"
513 });
514 }
515 }
516
517 // Clean up related DatasetDownload records to allow re-downloading
518 var relatedDownloads = await _context.DatasetDownloads
519 .Where(d => d.TaxonomyId == id || d.AccessionId == assembly.AccessionId)
520 .ToListAsync();
521
522 var removedDownloadCount = 0;
523 if (relatedDownloads.Any())
524 {
525 removedDownloadCount = relatedDownloads.Count;
526 _logger.LogInformation("Found {Count} related download records for assembly {AccessionId}",
527 removedDownloadCount, assembly.AccessionId);
528
529 _context.DatasetDownloads.RemoveRange(relatedDownloads);
530 _logger.LogInformation("Removed {Count} DatasetDownload records to allow re-downloading",
531 removedDownloadCount);
532 }
533
534 // Remove from database
535 _context.Assemblies.Remove(assembly);
536 await _context.SaveChangesAsync();
537
538 _logger.LogInformation("Assembly {TaxonomyId} ({OrganismName}) deleted successfully. Freed {Size} bytes. Removed {DownloadCount} download records.",
539 id, assembly.OrganismName, deletedSize, removedDownloadCount);
540
541 return Json(new {
542 success = true,
543 message = $"Assembly '{assembly.OrganismName}' deleted successfully. Freed {FormatFileSize(deletedSize)}. Dataset can now be re-downloaded if needed.",
544 deletedSize = deletedSize,
545 removedDownloads = removedDownloadCount
546 });
547 }
548 catch (Exception ex)
549 {
550 _logger.LogError(ex, "Error occurred while deleting assembly with ID: {TaxonomyId}", id);
551 return Json(new { success = false, message = "An error occurred while deleting the assembly" });
552 }
553 }

560 private long CalculateAssemblySize(string path)
561 {
562 if (string.IsNullOrEmpty(path) || !Directory.Exists(path))
563 return 0;
564
565 try
566 {
567 var directoryInfo = new DirectoryInfo(path);
568 return directoryInfo.GetFiles("*", SearchOption.AllDirectories)
569 .Sum(file => file.Length);
570 }
571 catch (Exception ex)
572 {
573 _logger.LogWarning(ex, "Failed to calculate size for directory: {Path}", path);
574 return 0;
575 }
576 }

583 private static string FormatFileSize(long bytes)
584 {
585 if (bytes == 0) return "0 B";
586
587 string[] sizes = { "B", "KB", "MB", "GB", "TB" };
588 int order = 0;
589 double size = bytes;
590
591 while (size >= 1024 && order < sizes.Length - 1)
592 {
593 order++;
594 size /= 1024;
595 }
596
597 return $"{size:0.##} {sizes[order]}";
598 }

605 private static string ExpandPath(string path)
606 {
607 if (string.IsNullOrEmpty(path))
608 return path;
609
610 // Expand ~ to home directory

612 {
613 var homeDir = Environment.GetFolderPath(Environment.SpecialFolder.UserProfile);
614 if (path == "~")
615 return homeDir;
616 return Path.Combine(homeDir, path.Substring(2));
617 }
618
619 // Expand environment variables
620 return Environment.ExpandEnvironmentVariables(path);

629 private string DetermineAssemblyType(string assemblyPath, string accessionId)
630 {
631 if (string.IsNullOrEmpty(assemblyPath))
632 {
633 return "Unknown";
634 }
635
636 try
637 {
638 // If directory doesn't exist, try to infer from accession pattern
639 if (!Directory.Exists(assemblyPath))
640 {
641 _logger.LogDebug("Assembly directory does not exist: {Path}, using accession-based detection", assemblyPath);
642 return DetermineTypeFromAccession(accessionId);
643 }
644
645 var files = Directory.GetFiles(assemblyPath, "*", SearchOption.TopDirectoryOnly)
646 .Select(f => Path.GetFileName(f).ToLowerInvariant())
647 .ToList();
648
649 _logger.LogDebug("Found {FileCount} files in {Path}: {Files}",
650 files.Count, assemblyPath, string.Join(", ", files.Take(5)));
651
652 if (files.Count == 0)
653 {
654 return "Empty";
655 }
656
657 // Check for different types of BLAST databases
658 var hasGenome = files.Any(f => f.Contains("genomic") && (f.EndsWith(".nhr") || f.EndsWith(".phr")));
659 var hasProtein = files.Any(f => f.Contains("protein") && f.EndsWith(".phr"));
660 var hasRNA = files.Any(f => f.Contains("rna") && f.EndsWith(".nhr"));
661 var hasCDS = files.Any(f => f.Contains("cds") && f.EndsWith(".nhr"));
662
663 _logger.LogDebug("Assembly type detection for {AccessionId}: Genome={Genome}, Protein={Protein}, RNA={RNA}, CDS={CDS}",
664 accessionId, hasGenome, hasProtein, hasRNA, hasCDS);
665
666 // Determine primary type based on available databases
667 var types = new List<string>();
668
669 if (hasGenome) types.Add("Genome");
670 if (hasProtein) types.Add("Proteome");
671 if (hasRNA) types.Add("Transcriptome");
672 if (hasCDS) types.Add("CDS");
673
674 if (types.Count == 0)
675 {
676 // Fallback: check for any BLAST database files
677 var hasAnyBlastDb = files.Any(f => f.EndsWith(".nhr") || f.EndsWith(".phr") ||
678 f.EndsWith(".nin") || f.EndsWith(".pin"));
679
680 if (hasAnyBlastDb)
681 {
682 return "Assembly";
683 }
684
685 // Check for FASTA files as another fallback
686 var hasFasta = files.Any(f => f.EndsWith(".fna") || f.EndsWith(".faa") ||
687 f.EndsWith(".fasta") || f.EndsWith(".fa"));
688 return hasFasta ? "FASTA" : "Unknown";
689 }
690
691 // Return combined type or primary type
692 if (types.Count == 1)
693 {
694 return types[0];
695 }
696 else if (types.Contains("Genome"))
697 {
698 // If genome is present with others, it's likely a complete genome assembly
699 return "Complete Genome";
700 }
701 else
702 {
703 // Multiple types without genome
704 return string.Join(" + ", types);
705 }
706 }
707 catch (Exception ex)
708 {
709 _logger.LogWarning(ex, "Failed to determine assembly type for path: {Path}", assemblyPath);
710 return "Error";
711 }
712 }

719 private string DetermineTypeFromAccession(string accessionId)
720 {
721 if (string.IsNullOrEmpty(accessionId))
722 {
723 return "Unknown";
724 }
725
726 // NCBI accession patterns
727 if (accessionId.StartsWith("GCF_") || accessionId.StartsWith("GCA_"))
728 {
729 return "Genome Assembly";
730 }
731 else if (accessionId.StartsWith("NC_"))
732 {
733 return "RefSeq Chromosome";
734 }
735 else if (accessionId.StartsWith("NM_"))
736 {
737 return "mRNA";
738 }
739 else if (accessionId.StartsWith("NP_"))
740 {
741 return "Protein";
742 }
743
744 return "Assembly";
745 }
