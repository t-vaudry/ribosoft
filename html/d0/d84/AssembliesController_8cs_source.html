97 [HttpPost]
98 public async Task<IActionResult> ToggleEnabled(int id)
99 {
100 _logger.LogInformation("Toggle enabled status for assembly with taxonomy ID: {TaxonomyId} by user: {User}", id, User.Identity?.Name);
101
102 try
103 {
104 var assembly = await _context.Assemblies.FirstOrDefaultAsync(a => a.TaxonomyId == id);
105 if (assembly == null)
106 {
107 return Json(new { success = false, message = "Assembly not found" });
108 }
109
110 assembly.IsEnabled = !assembly.IsEnabled;
111 await _context.SaveChangesAsync();
112
113 _logger.LogInformation("Assembly {TaxonomyId} enabled status changed to: {IsEnabled}", id, assembly.IsEnabled);
114
115 return Json(new {
116 success = true,
117 isEnabled = assembly.IsEnabled,
118 message = $"Assembly {(assembly.IsEnabled ? "enabled" : "disabled")} successfully"
119 });
120 }
121 catch (Exception ex)
122 {
123 _logger.LogError(ex, "Error occurred while toggling assembly enabled status for ID: {TaxonomyId}", id);
124 return Json(new { success = false, message = "An error occurred while updating the assembly status" });
125 }
