105 [AutomaticRetry(Attempts = 0)]
106 public async Task Phase1(int jobId, IJobCancellationToken cancellationToken)
107 {
108 var job = GetJob(jobId);
109
110 // Log job execution start
111 await _activityLogService.LogJobExecutionAsync(
112 message: $"Job {jobId} Phase1 started - Candidate generation and structure calculation",
113 jobId: jobId,
114 userId: job.OwnerId,
115 userName: job.Owner?.UserName
116 );
117
118 // TODO - temporarily catch retried jobs
119 await DoStage(job, JobState.Errored, j => j.JobState != JobState.New, async (j, c) => { await Task.CompletedTask; }, cancellationToken);
120
121 // run candidate generator
122 await DoStage(job, JobState.CandidateGenerator, j => j.JobState == JobState.New, RunCandidateGenerator, cancellationToken);
123
124 // calculate structure score
125 await DoStage(job, JobState.Structure, j => j.JobState == JobState.CandidateGenerator, CalculateStructure, cancellationToken);
126
127 // After structure calculation, queue the appropriate next phase
128 // Reload job to get current state after structure calculation
129 job = GetJob(jobId);
130
131 if (job.JobState == JobState.Structure)
132 {
133 if (job.TargetEnvironment == TargetEnvironment.InVivo)
134 {
135 // InVivo jobs need BLAST analysis (Phase2)
136 await UpdateJobProperties(jobId, JobState.QueuedPhase2, "Queued for BLAST analysis");
137
138 await _activityLogService.LogJobExecutionAsync(
139 message: $"Job {jobId} Phase1 completed - Queued for Phase2 (BLAST analysis)",
140 jobId: jobId,
141 userId: job.OwnerId,
142 userName: job.Owner?.UserName
143 );
144
145 BackgroundJob.Enqueue<GenerateCandidates>(x => x.Phase2(jobId, JobCancellationToken.Null));
146 }
147 else if (job.TargetEnvironment == TargetEnvironment.InVitro)
148 {
149 // InVitro jobs skip BLAST and go directly to optimization (Phase3)
150 await UpdateJobProperties(jobId, JobState.QueuedPhase3, "Queued for multi-objective optimization");
151 BackgroundJob.Enqueue<GenerateCandidates>(x => x.Phase3(jobId, JobCancellationToken.Null));
152 }
153 }

165 public async Task Phase2(int jobId, IJobCancellationToken cancellationToken)
166 {
167 var job = GetJob(jobId);
168

174

179 await Task.CompletedTask;

212 private async Task DoStage(Job job, JobState state, Func<Job, bool> acceptFunc, Func<Job, IJobCancellationToken, Task> func, IJobCancellationToken cancellationToken)
213 {
214 if (!acceptFunc(job))
215 {
216 // don't do anything if the stage can't handle this type of job
217 // this also catches errored jobs, etc.
218 return;
219 }
220
221 cancellationToken.ThrowIfCancellationRequested();
222
223 // set the job to this stage's state
224 if (job.JobState != state)
225 {
226 var statusMessage = GetStatusMessageForState(state);
227 await UpdateJobProperties(job.Id, state, statusMessage);
228 }
229
230 await func(job, cancellationToken);
231 }

238 private string GetStatusMessageForState(JobState state)
239 {
240 return state switch
241 {
242 JobState.New => "Job created and queued for processing",
243 JobState.CandidateGenerator => "Generating ribozyme candidates...",
244 JobState.Structure => "Calculating structure scores...",
245 JobState.Specificity => "Running BLAST analysis for specificity...",
246 JobState.MultiObjectiveOptimization => "Optimizing and ranking candidates...",
247 JobState.QueuedPhase2 => "Queued for BLAST analysis (Phase 2)",
248 JobState.QueuedPhase3 => "Queued for optimization (Phase 3)",
249 JobState.Completed => "Job completed successfully",
250 JobState.Warning => "Job completed with warnings",
251 JobState.Errored => "Job failed with errors",
252 JobState.Cancelled => "Job was cancelled",
253 _ => $"Job in state: {state}"
254 };

394 private async Task RunCandidateGenerator(Job job, IJobCancellationToken cancellationToken)
395 {
396 List<string> rnaInputs = new List<string>();
397 if (job.FivePrime || job.OpenReadingFrame || job.ThreePrime)
398 {
399 SetTargetRegions(job, ref rnaInputs);
400 }
401 else
402 {
403 await UpdateJobProperties(job.Id, JobState.Warning, "No Target Region Selected!");
404 return;
405 }
406
407 CandidateGeneration.CandidateGenerator candidateGenerator = new CandidateGeneration.CandidateGenerator();
408 foreach (var rnaInput in rnaInputs)
409 {
410 RNAStructure = _ribosoftAlgo.MFEFold(rnaInput);
411
412 foreach (var ribozymeStructure in job.Ribozyme?.RibozymeStructures ?? new List<RibozymeStructure>())
413 {
414 cancellationToken.ThrowIfCancellationRequested();
415 IEnumerable<Candidate> candidates;
416 try
417 {
418 // Candidate Generation
419 candidates = candidateGenerator.GenerateCandidates(
420 ribozymeStructure.Sequence,
421 ribozymeStructure.Structure,
422 ribozymeStructure.SubstrateTemplate,
423 ribozymeStructure.SubstrateStructure,
424 rnaInput);
425 }
426 catch (CandidateGeneration.CandidateGenerationException e)
427 {
428 _logger.LogError(e, "Exception occurred during Candidate Generation.");
429 await UpdateJobProperties(job.Id, JobState.Errored, e.Message);
430 return;
431 }
432
433 // Algorithms
434 try
435 {
436 uint batchCount = 0;
437 uint totalProcessed = 0;
438 uint totalCandidates = (uint)candidates.Count();
439 _db.ChangeTracker.AutoDetectChangesEnabled = false;
440
441 foreach (var candidate in candidates)
442 {
443 cancellationToken.ThrowIfCancellationRequested();
444 RunScoreAlgorithms(candidate, job, ribozymeStructure, RNAStructure);
445 totalProcessed++;
446
447 if (++batchCount % 100 == 0)
448 {
449 await _db.SaveChangesAsync();
450 await RecreateDbContext();
451 _db.ChangeTracker.AutoDetectChangesEnabled = false;
452 batchCount = 0;
453
454 // Update progress status
455 var progressMessage = $"Processing candidates: {totalProcessed}/{totalCandidates} completed";
456 await UpdateJobProperties(job.Id, JobState.CandidateGenerator, progressMessage);
457 }
458 }
459
460 await RecreateDbContext();
461 }
462 catch (RibosoftAlgoException e)
463 {
464 _logger.LogError(e, "Exception occurred during Ribosoft Algorithms.");
465 await UpdateJobProperties(job.Id, JobState.Errored, e.Code.ToString());
466 return;
467 }
468 finally
469 {
470 _db.ChangeTracker.AutoDetectChangesEnabled = true;
471 await UpdateJobProperties(job.Id, job.JobState, job.StatusMessage);
472 }
473
474 candidateGenerator.Clear();
475 }
476 }
477
478 _db.ChangeTracker.AutoDetectChangesEnabled = false;
479 var designs = _db.Designs.Where(j => j.JobId == job.Id).ToList();
480
481 // Check that there are designs left
482 if (!designs.Any())
483 {
484 _db.ChangeTracker.AutoDetectChangesEnabled = true;
485 await UpdateJobProperties(job.Id, JobState.Warning, "No designs returned from Candidate Generation!");
486 _logger.LogError("No designs returned from Candidate Generation!");
487 return;
488 }
489
490 var maxDesiredTemp = designs.Max(d => d.DesiredTemperatureScore.GetValueOrDefault());
491 var minDesiredTemp = designs.Min(d => d.DesiredTemperatureScore.GetValueOrDefault());
492 var maxAccessibility = designs.Max(d => d.AccessibilityScore.GetValueOrDefault());
493 var minAccessibility = designs.Min(d => d.AccessibilityScore.GetValueOrDefault());
494
495 var newDesiredTempTolerance = job.DesiredTempTolerance * (maxDesiredTemp - minDesiredTemp);
496 var newAccessibilityTolerance = job.AccessibilityTolerance * (maxAccessibility - minAccessibility);
497
498 _db.ChangeTracker.AutoDetectChangesEnabled = true;
499 await UpdateJobTolerances(job.Id, newDesiredTempTolerance, newAccessibilityTolerance);
500
501 // Save all designs to database
502 var designCount = designs.Count();
503 }

510 private void SetTargetRegions(Job job, ref List<string> rnaInputs)
511 {
512 if (job.FivePrime && job.OpenReadingFrame && job.ThreePrime)

514 rnaInputs.Add(job.RNAInput);

521 {
522 rnaInputs.Add(job.RNAInput.Substring(job.OpenReadingFrameStart, job.RNAInput.Length - job.OpenReadingFrameStart - 1));

530 {
531 rnaInputs.Add(job.RNAInput.Substring(0, job.OpenReadingFrameStart));

533 else if (job.OpenReadingFrame)
534 {
535 rnaInputs.Add(job.RNAInput.Substring(job.OpenReadingFrameStart, job.OpenReadingFrameEnd - job.OpenReadingFrameStart - 1));
536 }
537 else if (job.ThreePrime)
538 {
539 rnaInputs.Add(job.RNAInput.Substring(job.OpenReadingFrameEnd, job.RNAInput.Length - job.OpenReadingFrameEnd - 1));

549 private void RunScoreAlgorithms(Candidate candidate, Job job, RibozymeStructure ribozymeStructure, string RNAStructure)
550 {
551 var idealStructurePattern = new Regex(@"[^.^(^)]");
552 string ideal = idealStructurePattern.Replace(candidate.Structure ?? string.Empty, ".");
553
554 float naConcentration = job.Na.GetValueOrDefault();
555 float probeConcentration = job.Probe.GetValueOrDefault();
556 float targetTemperature = job.TargetTemperature.GetValueOrDefault();
557
558 var temperatureScore = _ribosoftAlgo.Anneal(candidate, candidate.SubstrateSequence ?? string.Empty,
559 candidate.SubstrateStructure ?? string.Empty, naConcentration, probeConcentration, targetTemperature);
560
561 foreach (var cutsiteIndex in candidate.CutsiteIndices ?? new List<int>())
562 {
563 var accessibilityScore = _ribosoftAlgo.Accessibility(candidate, RNAStructure,
564 cutsiteIndex, naConcentration, probeConcentration, targetTemperature);
565
566 _db.Designs.Add(new Design
567 {
568 JobId = job.Id,
569
570 Sequence = candidate.Sequence?.GetString() ?? string.Empty,
571 IdealStructure = ideal,
572 SubstrateSequence = candidate.SubstrateSequence ?? "",
573
574 // TODO: save actual cutsite (cutsiteIndex + ribozymeStructure.Cutsite + candidate.CutsiteNumberOffset)
575 CutsiteIndex = cutsiteIndex,
576
577 SubstrateSequenceLength = candidate.SubstrateSequence?.Length ?? 0,
578 AccessibilityScore = accessibilityScore,
579 DesiredTemperatureScore = temperatureScore
580 });
581 }
582 }

631 private async Task RunBlast(Job job, IJobCancellationToken cancellationToken)
632 {
633 // check if blastn is available; if it isn't, ignore specificity
634 if (!_blaster.IsAvailable())
635 {
636 _logger.LogWarning("RibosoftWarning | BLAST Service is not available!!");
637 return;
638 }
639
640 var designs = _db.Designs
641 .Where(d => d.JobId == job.Id)
642 .AsEnumerable()
643 .GroupBy(d => new { d.CutsiteIndex, d.SubstrateSequence });
644
645 foreach (var designGroup in designs)
646 {
647 cancellationToken.ThrowIfCancellationRequested();
648
649 var design = designGroup.First();
650 var substrateSequence = design.SubstrateSequence;
651
652 if (string.IsNullOrEmpty(substrateSequence))
653 {
654 continue;
655 }
656
657 // calculate the substrate specificity score, which is common to all designs in this group
658 var substrateSpecificityScore = CalculateSpecificity(substrateSequence, job.Assembly?.Path ?? "");
659
660 foreach (var d in designGroup)
661 {
662 d.SpecificityScore = substrateSpecificityScore;
663 }
664 }
665
666 // Specificity is minimized to 1 in the case of a wildtype gene (a score of 0 is ideal for synthetic genes)
667 // Anything below 1 means there is absolutely no matching in the organism and will not bond
668 // Therefore, remove the design
669 if (job.SpecificityMethod == SpecificityMethod.Wildtype)
670 {
671 _db.Designs.RemoveRange(_db.Designs.Where(d => d.SpecificityScore < 1.0f));
672 }
673
674 var completedDesigns = _db.Designs.Where(d => d.JobId == job.Id);
675
676 // Check that there are designs left
677 if (!completedDesigns.Any())
678 {
679 await UpdateJobProperties(job.Id, JobState.Warning, "No designs returned from Candidate Generation!");
680 _logger.LogError("No designs returned from Candidate Generation!");
681 return;
682 }
683
684 float deltaSpecificity = completedDesigns.Max(d => d.SpecificityScore.GetValueOrDefault()) - completedDesigns.Min(d => d.SpecificityScore.GetValueOrDefault());
685 var newSpecificityTolerance = job.SpecificityTolerance * deltaSpecificity;
686 await UpdateJobSpecificityTolerance(job.Id, newSpecificityTolerance);
687 }

695 private float CalculateSpecificity(string sequence, string database)
696 {
697 var specificityScore = 0.0f;
698 var blastParameters = BlastParametersForQuery(database, sequence);
699 blastParameters.Query = sequence;
700 var output = _blaster.Run(blastParameters);
701
702 foreach (var line in output.Split(new[] {Environment.NewLine}, StringSplitOptions.RemoveEmptyEntries))
703 {
704 var fields = line.Split('\t');
705
706 // filter out X_ predictions
707 //if (fields[1].StartsWith('X'))
708 //{
709 // continue;
710 //}
711
712 specificityScore += float.Parse(fields[2]) * float.Parse(fields[3]) / 10000;
713 }
714
715 return specificityScore;
716 }

724 private BlastParameters BlastParametersForQuery(string database, string query)
725 {
726 var blastParameters = new BlastParameters
727 {
728 BlastDbPath = _configuration.GetValue("Blast:BLASTDB", string.Empty) ?? string.Empty,
729 Database = database,
730 UseIndex = true,
731 LowercaseMasking = true,
732 OutputFormat = "6 qseqid saccver pident qcovs",
733 NumThreads = _configuration.GetValue("Blast:NumThreads", 4),
734 MaxTargetSequences = 200
735 };
736
737 if (query.Length <= 30)
738 {
739 // adjust parameters for a short query (as NCBI does it)
740 blastParameters.Task = BlastParameters.BlastTask.blastn_short;
741 blastParameters.ExpectValue = 1000.0f;
742 }
743
744 return blastParameters;
