724 [HttpGet]
725 public async Task<IActionResult> GetRecoveryCodes()
726 {
727 var user = await _userManager.GetUserAsync(User);
728 if (user == null)
729 {
730 return Json(new { success = false, error = "User not found" });
731 }
732
733 if (!user.TwoFactorEnabled)
734 {
735 return Json(new { success = false, error = "Two-factor authentication is not enabled" });
736 }
737
738 try
739 {
740 // Get the count of remaining recovery codes
741 var recoveryCodesLeft = await _userManager.CountRecoveryCodesAsync(user);
742
743 if (recoveryCodesLeft == 0)
744 {
745 return Json(new {
746 success = false,
747 error = "No recovery codes available. Please generate new codes.",
748 needsGeneration = true
749 });
750 }
751
752 // Note: ASP.NET Core Identity doesn't provide a way to retrieve the actual
753 // recovery codes for security reasons. They are hashed and stored securely.
754 // We can only show the count and suggest generating new ones.
755 return Json(new {
756 success = false,
757 error = $"You have {recoveryCodesLeft} recovery codes remaining, but the actual codes cannot be displayed for security reasons. You can generate new codes if needed.",
758 recoveryCodesLeft = recoveryCodesLeft,
759 cannotDisplay = true
760 });
761 }
762 catch (Exception ex)
763 {
764 _logger.LogError(ex, "Error retrieving recovery codes for user {UserId}", user.Id);
765 return Json(new { success = false, error = "Error retrieving recovery codes" });
766 }

786 private string FormatKey(string unformattedKey)
787 {
788 var result = new StringBuilder();
789 int currentPosition = 0;
790 while (currentPosition + 4 < unformattedKey.Length)
791 {
792 result.Append(unformattedKey.Substring(currentPosition, 4)).Append(" ");
793 currentPosition += 4;
794 }
795 if (currentPosition < unformattedKey.Length)
796 {
797 result.Append(unformattedKey.Substring(currentPosition));
798 }
799
800 return result.ToString().ToLowerInvariant();
801 }
