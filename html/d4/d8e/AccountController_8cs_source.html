102 [HttpPost]
103 [AllowAnonymous]
104 [ValidateAntiForgeryToken]
105 public async Task<IActionResult> Login(LoginViewModel model, string? returnUrl = null)
106 {
107 ViewData["ReturnUrl"] = returnUrl;
108 if (ModelState.IsValid)
109 {
110 // This doesn't count login failures towards account lockout
111 // To enable password failures to trigger account lockout, set lockoutOnFailure: true
112 var result = await _signInManager.PasswordSignInAsync(model.Email, model.Password, model.RememberMe, lockoutOnFailure: false);
113 if (result.Succeeded)
114 {
115 var user = await _userManager.FindByEmailAsync(model.Email);
116 _logger.LogInformation("User logged in.");
117
118 // Log successful authentication
119 await _activityLogService.LogAuthenticationAsync(
120 message: $"User '{model.Email}' logged in successfully",
121 userId: user?.Id,
122 userName: user?.UserName,
123 ipAddress: HttpContext.Connection.RemoteIpAddress?.ToString(),
124 success: true
125 );
126
127 return RedirectToLocal(returnUrl);
128 }
129 if (result.RequiresTwoFactor)
130 {
131 return RedirectToAction(nameof(LoginWith2fa), new { returnUrl, model.RememberMe });
132 }
133 else
134 {
135 // Log failed authentication
136 await _activityLogService.LogAuthenticationAsync(
137 message: $"Failed login attempt for '{model.Email}'",
138 userName: model.Email,
139 ipAddress: HttpContext.Connection.RemoteIpAddress?.ToString(),
140 success: false
141 );
142
143 ModelState.AddModelError(string.Empty, "Invalid login attempt.");
144 return View(model);
145 }
146 }
147
148 // If we got this far, something failed, redisplay form
149 return View(model);
150 }

279 [HttpPost]
280 [AllowAnonymous]
281 [ValidateAntiForgeryToken]
282 public async Task<IActionResult> LoginWithRecoveryCode(LoginWithRecoveryCodeViewModel model, string? returnUrl = null)
283 {
284 if (!ModelState.IsValid)
285 {
286 return View(model);
287 }
288
289 var user = await _signInManager.GetTwoFactorAuthenticationUserAsync();
290 if (user == null)
291 {
292 throw new ApplicationException($"Unable to load two-factor authentication user.");
293 }
294
295 // Validate the one-time code
296 var isValidCode = await _oneTimeCodeService.ValidateCodeAsync(user.Id, model.OneTimeCode, "2FA_BYPASS");
297
298 if (isValidCode)
299 {
300 // Sign in the user (similar to 2FA recovery code sign-in)
301 await _signInManager.SignInAsync(user, isPersistent: false);
302
303 _logger.LogInformation("User with ID {UserId} logged in with a one-time email code.", user.Id);
304 return RedirectToLocal(returnUrl);
305 }
306 else
307 {
308 _logger.LogWarning("Invalid one-time code entered for user with ID {UserId}", user.Id);
309 ModelState.AddModelError(string.Empty, "Invalid or expired verification code. Please try again or request a new code.");
310
311 // Prepare model for re-display
312 model.Email = user.Email!;
313 model.CodeSent = true;
314 model.CanResend = true;
315 model.ResendCooldownSeconds = 0;
316
317 return View(model);
318 }

333 {

335 }
336
337 try

340 _logger.LogInformation("One-time code resent for user with ID {UserId}", user.Id);
341
342 return Json(new { success = true, message = "A new verification code has been sent to your email." });

344 catch (Exception ex)

346 _logger.LogError(ex, "Failed to resend one-time code for user with ID {UserId}", user.Id);

482 // For more information on how to enable account confirmation and password reset please

484 var code = await _userManager.GeneratePasswordResetTokenAsync(user);
485 var callbackUrl = Url.ResetPasswordCallbackLink(user.Id, code, Request.Scheme);
486 await _emailSender.SendPasswordResetAsync(model.Email, callbackUrl);
487 return RedirectToAction(nameof(ForgotPasswordConfirmation));
