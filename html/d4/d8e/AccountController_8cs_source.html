102 {
103 // This doesn't count login failures towards account lockout
104 // To enable password failures to trigger account lockout, set lockoutOnFailure: true
105 var result = await _signInManager.PasswordSignInAsync(model.Email, model.Password, model.RememberMe, lockoutOnFailure: false);
106 if (result.Succeeded)
107 {
108 _logger.LogInformation("User logged in.");
109 return RedirectToLocal(returnUrl);
110 }
111 if (result.RequiresTwoFactor)
112 {
113 return RedirectToAction(nameof(LoginWith2fa), new { returnUrl, model.RememberMe });
114 }
115 else
116 {
117 ModelState.AddModelError(string.Empty, "Invalid login attempt.");
118 return View(model);

184 [AllowAnonymous]
185 [ValidateAntiForgeryToken]
186 public async Task<IActionResult> LoginWith2fa(LoginWith2faViewModel model, bool rememberMe, string? returnUrl = null)
187 {
188 if (!ModelState.IsValid)
189 {
190 return View(model);
191 }
192
193 var user = await _signInManager.GetTwoFactorAuthenticationUserAsync();
194 if (user == null)
195 {
196 throw new ApplicationException($"Unable to load user with ID '{_userManager.GetUserId(User)}'.");
197 }
198
199 var authenticatorCode = model.TwoFactorCode.Replace(" ", string.Empty).Replace("-", string.Empty);
200
201 var result = await _signInManager.TwoFactorAuthenticatorSignInAsync(authenticatorCode, rememberMe, model.RememberMachine);
202
203 if (result.Succeeded)
204 {
205 _logger.LogInformation("User with ID {UserId} logged in with 2fa.", user.Id);
206 return RedirectToLocal(returnUrl);
207 }
208 else
209 {
210 _logger.LogWarning("Invalid authenticator code entered for user with ID {UserId}.", user.Id);
211 ModelState.AddModelError(string.Empty, "Invalid authenticator code.");
212 return View();

256 public async Task<IActionResult> LoginWithRecoveryCode(LoginWithRecoveryCodeViewModel model, string? returnUrl = null)
257 {
258 if (!ModelState.IsValid)
259 {
260 return View(model);

263 var user = await _signInManager.GetTwoFactorAuthenticationUserAsync();
264 if (user == null)
265 {
266 throw new ApplicationException($"Unable to load two-factor authentication user.");

274 // Sign in the user (similar to 2FA recovery code sign-in)
275 await _signInManager.SignInAsync(user, isPersistent: false);
276
277 _logger.LogInformation("User with ID {UserId} logged in with a one-time email code.", user.Id);
278 return RedirectToLocal(returnUrl);
279 }
280 else
281 {
282 _logger.LogWarning("Invalid one-time code entered for user with ID {UserId}", user.Id);
283 ModelState.AddModelError(string.Empty, "Invalid or expired verification code. Please try again or request a new code.");
284
285 // Prepare model for re-display
286 model.Email = user.Email!;
287 model.CodeSent = true;
288 model.CanResend = true;
289 model.ResendCooldownSeconds = 0;
290
291 return View(model);

304 {
305 var user = await _signInManager.GetTwoFactorAuthenticationUserAsync();
306 if (user == null)
307 {
308 return Json(new { success = false, message = "Session expired. Please start over." });
309 }
310
311 try
312 {
313 await _oneTimeCodeService.GenerateAndSendCodeAsync(user.Id, user.Email!, "2FA_BYPASS");
314 _logger.LogInformation("One-time code resent for user with ID {UserId}", user.Id);
315
316 return Json(new { success = true, message = "A new verification code has been sent to your email." });
317 }
318 catch (Exception ex)
319 {
320 _logger.LogError(ex, "Failed to resend one-time code for user with ID {UserId}", user.Id);
321 return Json(new { success = false, message = "Failed to send verification code. Please try again." });
322 }

369 return View(new RegisterViewModel()); // Return empty model for clean form

434 {
435 if (ModelState.IsValid)
436 {
437 var user = await _userManager.FindByEmailAsync(model.Email);
438 if (user == null || !(await _userManager.IsEmailConfirmedAsync(user)))
439 {
440 // Don't reveal that the user does not exist or is not confirmed
441 return RedirectToAction(nameof(ForgotPasswordConfirmation));

443
444 // For more information on how to enable account confirmation and password reset please
445 // visit https://go.microsoft.com/fwlink/?LinkID=532713
446 var code = await _userManager.GeneratePasswordResetTokenAsync(user);
447 var callbackUrl = Url.ResetPasswordCallbackLink(user.Id, code, Request.Scheme);
448 await _emailSender.SendPasswordResetAsync(model.Email, callbackUrl);
449 return RedirectToAction(nameof(ForgotPasswordConfirmation));
450 }
