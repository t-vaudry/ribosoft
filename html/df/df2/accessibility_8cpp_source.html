34std::vector<AccessibleSection> find_accessible_sections(const std::string& arm_sequence, const std::string& arm_folded_structure) {
35 std::vector<AccessibleSection> sections;
36 int current_start = -1;
37
38 for (int i = 0; i < static_cast<int>(arm_sequence.length()); i++) {
39 if (arm_folded_structure[i] == '.') { // Accessible
40 if (current_start == -1) {
41 current_start = i; // Start new section
42 }
43 } else { // Bound/paired
44 if (current_start != -1) {
45 // End current section
46 sections.push_back({
47 current_start,

51 current_start = -1;

53 }

58 sections.push_back({

61 arm_sequence.substr(current_start)

63 }

76double calculate_section_accessibility_score(const std::vector<AccessibleSection>& sections,
77 float na_concentration, float probe_concentration, float target_temp) {
78 const int MIN_BINDING_LENGTH = 3; // Minimum length for effective binding
79 double total_score = 0.0;
80
81 for (const auto& section : sections) {
82 if (section.length < MIN_BINDING_LENGTH) {
83 continue; // Too short to be useful for binding
84 }
85
86 // Calculate melting temperature for this accessible section
87 // Lock needed as melting library is not thread-safe
88 std::lock_guard<std::mutex> lock(melting_mutex);
89 double section_melting_temp = melting(section.sequence.c_str(), na_concentration, probe_concentration);
90
91 // Calculate temperature difference penalty
92 double difference = fabs(section_melting_temp - target_temp);
93
94 // Apply scoring similar to anneal function
95 if (difference <= 4.0) {
96 total_score += difference;
97 } else {
98 total_score += pow(difference, 2);
99 }
100 }
101
102 return total_score;
