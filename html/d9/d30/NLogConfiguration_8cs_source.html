51 private static void ConfigureWithDatabase(string connectionString, string dbProvider, string commandText)
52 {
53 var config = new LoggingConfiguration();
54
55 // Load existing configuration from file
56 var fileConfig = new XmlLoggingConfiguration("nlog.config");
57
58 // Copy existing targets and rules (excluding database target)
59 foreach (var target in fileConfig.AllTargets)
60 {
61 if (target.Name != "database")
62 {
63 config.AddTarget(target);
64 }
65 }
66
67 foreach (var rule in fileConfig.LoggingRules)
68 {
69 if (!rule.Targets.Any(t => t.Name == "database"))
70 {
71 config.LoggingRules.Add(rule);
72 }
73 }
74
75 // Create and configure database target
76 var databaseTarget = CreateDatabaseTarget(connectionString, dbProvider, commandText);
77 config.AddTarget(databaseTarget);
78
79 // Add database logging rules
80 AddDatabaseLoggingRules(config, databaseTarget);
81
82 // Apply the configuration
83 LogManager.Configuration = config;

102 private static void AddDatabaseParameters(DatabaseTarget databaseTarget)
103 {
104 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@timestamp", "${date:universalTime=true:format=yyyy-MM-dd HH\\:mm\\:ss.fff zzz}"));
105
106 // Convert short log levels to full names to match ActivityLogService format
107 var levelLayout = "${replace:inner=${replace:inner=${replace:inner=${replace:inner=${level}:searchFor=Info:replaceWith=Information}:searchFor=Warn:replaceWith=Warning}:searchFor=Error:replaceWith=Error}:searchFor=Debug:replaceWith=Debug}";
108 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@level", levelLayout));
109
110 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@logger", "${logger}"));
111 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@message", "${message}"));
112 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@exception", "${exception:format=tostring}"));
113 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@username", "${aspnet-user-identity}"));
114 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@ipaddress", "${aspnet-request-ip}"));
115 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@requestpath", "${aspnet-request-url:IncludeHost=false:IncludePort=false:IncludeQueryString=false}"));
116 databaseTarget.Parameters.Add(new DatabaseParameterInfo("@requestmethod", "${aspnet-request-method}"));
